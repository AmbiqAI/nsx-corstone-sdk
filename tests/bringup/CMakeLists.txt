cmake_minimum_required(VERSION 3.24)
project(corstone_bringup LANGUAGES C)
set(NSX_BOARD corstone300_fvp_u55 CACHE STRING "FVP board to qualify")
if(NSX_BOARD STREQUAL "corstone320_fvp_u85")
    set(NSX_SOC_FAMILY corstone320)
else()
    set(NSX_SOC_FAMILY corstone300)
endif()
set(NSX_BSP_CONTRACT_REQUIRED 2)
set(root "${CMAKE_CURRENT_LIST_DIR}/../..")
include("${root}/cmake/nsx_corstone_bsp.cmake")
foreach(module sdk soc bsp)
    add_subdirectory("${root}/modules/nsx-corstone-${module}" "${module}")
endforeach()
add_executable(bringup main.c)
target_link_libraries(bringup PRIVATE nsx::bsp)

option(TEST_NPU "Link the NPU initialization path in the cross-build" OFF)
if(TEST_NPU)
    add_subdirectory("${root}/modules/nsx-corstone-npu" npu)
    target_link_libraries(bringup PRIVATE nsx::corstone_npu)
    target_compile_definitions(bringup PRIVATE TEST_NPU)
endif()
